# MS-03.01-C: Database Schema Design (D.E.C.I.D.E - Execute)

## Objective
Define Firestore data model, create TypeScript interfaces, implement Zod validation schemas.

**Core requirement:** Type-safe, validated data contracts.

## Firestore Data Model

### Collection: `users`
**Document ID:** Session UUID (localStorage) or Auth UID (future)

**Schema:**
```typescript
interface UserPreferenceDocument {
  userId: string;
  language: 'en' | 'es';
  favorites: string[];
  createdAt: Timestamp;
  updatedAt: Timestamp;
}
```

## TypeScript Interfaces

**File: `types/user-preference.ts`**
```typescript
export interface UserPreferenceInput {
  userId: string;
  language: 'en' | 'es';
  favorites?: string[];
}

export interface UserPreferenceResponse {
  userId: string;
  language: 'en' | 'es';
  favorites: string[];
  createdAt: string; // ISO 8601
  updatedAt: string;
}
```

## Zod Validation

**File: `lib/schemas/user-preference.schema.ts`**
```typescript
import { z } from 'zod';

export const UserPreferenceInputSchema = z.object({
  userId: z.string().uuid(),
  language: z.enum(['en', 'es']),
  favorites: z.array(z.string()).default([]).optional(),
});

export type UserPreferenceInput = z.infer<typeof UserPreferenceInputSchema>;
```

## Helper Functions

**File: `lib/user-preferences.ts`**
```typescript
import { firestoreDb } from './firebase';
import { UserPreferenceInputSchema } from './schemas/user-preference.schema';

export async function getUserPreference(userId: string) {
  const db = firestoreDb();
  const doc = await db.collection('users').doc(userId).get();

  if (!doc.exists) return null;

  const data = doc.data();
  return {
    ...data,
    createdAt: data.createdAt.toDate().toISOString(),
    updatedAt: data.updatedAt.toDate().toISOString(),
  };
}

export async function setUserPreference(input: UserPreferenceInput) {
  const validated = UserPreferenceInputSchema.parse(input);
  const db = firestoreDb();

  await db.collection('users').doc(validated.userId).set({
    ...validated,
    updatedAt: Timestamp.now(),
  });

  return getUserPreference(validated.userId);
}
```
