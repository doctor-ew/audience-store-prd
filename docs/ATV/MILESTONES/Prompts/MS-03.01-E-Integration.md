# MS-03.01-E: Frontend Integration & Migration (D.E.C.I.D.E - Execute)

## Objective
Refactor `store/usePreferences.ts` to use `/api/preferences` instead of localStorage, implement UUID session management, migrate existing preferences.

**Core requirement:** Seamless UX—language preference persists across devices.

## Migration Strategy
1. Generate/retrieve session UUID from localStorage
2. Migrate existing language preference to Firestore
3. Use Firestore as source of truth, localStorage as fallback

## Implementation

**File: `store/usePreferences.ts` (refactor)**
```typescript
import { useState, useEffect } from 'react';
import { v4 as uuidv4 } from 'uuid';
import { useTranslation } from 'react-i18next';

export function usePreferences() {
  const { i18n } = useTranslation();
  const [userId, setUserId] = useState('');
  const [preference, setPreference] = useState(null);
  const [isLoading, setIsLoading] = useState(true);

  // Step 1: Initialize UUID
  useEffect(() => {
    let id = localStorage.getItem('userId');
    if (!id) {
      id = uuidv4();
      localStorage.setItem('userId', id);
    }
    setUserId(id);
  }, []);

  // Step 2: Fetch from Firestore or migrate
  useEffect(() => {
    if (!userId) return;

    async function loadPreference() {
      try {
        const res = await fetch(`/api/preferences?userId=${userId}`);

        if (res.ok) {
          const data = await res.json();
          setPreference(data);
          i18n.changeLanguage(data.language);
        } else if (res.status === 404) {
          // Migrate from localStorage
          const localLang = localStorage.getItem('i18nextLng') || 'en';

          await fetch('/api/preferences', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ userId, language: localLang, favorites: [] }),
          });
        }

        setIsLoading(false);
      } catch (error) {
        // Fallback to localStorage
        const localLang = localStorage.getItem('i18nextLng') || 'en';
        setPreference({ userId, language: localLang, favorites: [] });
        setIsLoading(false);
      }
    }

    loadPreference();
  }, [userId, i18n]);

  // Step 3: Update language
  const updateLanguage = async (language: 'en' | 'es') => {
    try {
      await fetch('/api/preferences', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ userId, language, favorites: preference.favorites }),
      });

      i18n.changeLanguage(language);
      document.documentElement.lang = language;
    } catch (error) {
      // Fallback: Update locally
      i18n.changeLanguage(language);
      localStorage.setItem('i18nextLng', language);
    }
  };

  return { preference, isLoading, updateLanguage };
}
```
